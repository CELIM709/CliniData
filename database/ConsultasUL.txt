-- ----------------------------------------------------------------------------
-- Pregunta 1: ¿Qué citas tiene programadas un paciente?
-- 
-- Explicación Técnica:
-- 1. Realiza una combinación de tablas (JOIN) entre 'cita', 'paciente' y 'persona' 
--    para obtener la información del paciente y su agenda.
-- 2. Une adicionalmente con 'medico' y 'persona' para conocer el médico tratante.
-- 3. Aplica un filtro en la cláusula WHERE especificando la cédula del paciente 
--    y validando que el estado de la cita sea 'PROGRAMADA' (o distinto de 'CANCELADA').
-- ----------------------------------------------------------------------------
SELECT 
    c.id_cita,
    c.rango_cita,
    c.consultorio,
    c.estado,
    p_med.nombre AS nombre_medico,
    p_med.apellido AS apellido_medico
FROM cita c
JOIN paciente pac ON c.cedula_paciente = pac.cedula
JOIN medico m ON c.cedula_medico = m.cedula
JOIN persona p_med ON m.cedula = p_med.cedula
WHERE c.cedula_paciente = 'V-22333444' -- Parámetro: Cédula del paciente
  AND c.estado = 'PROGRAMADA';


-- ----------------------------------------------------------------------------
-- Pregunta 2: ¿Qué pacientes ha atendido un médico en una fecha determinada?
-- 
-- Explicación Técnica:
-- 1. Relaciona la tabla 'consulta' con 'paciente' y 'persona' para extraer la 
--    identificación y los datos filiatorios de los pacientes atendidos.
-- 2. Utiliza la función DATE() sobre el campo 'fecha' de la consulta para comparar 
--    únicamente el componente de fecha (año-mes-día) e ignorar la hora.
-- 3. Filtra mediante la cédula del médico tratante y la fecha específica requerida.
-- ----------------------------------------------------------------------------
SELECT DISTINCT
    p_pac.cedula,
    p_pac.nombre,
    p_pac.apellido,
    p_pac.telefono,
    p_pac.email
FROM consulta con
JOIN paciente pac ON con.cedula_paciente = pac.cedula
JOIN persona p_pac ON pac.cedula = p_pac.cedula
WHERE con.cedula_medico = 'V-15234567' -- Parámetro: Cédula del médico
  AND DATE(con.fecha) = '2026-08-25'; -- Parámetro: Fecha a consultar


-- ----------------------------------------------------------------------------
-- Pregunta 3: ¿Qué estudios tiene pendientes un paciente?
-- 
-- Explicación Técnica:
-- 1. Combina las tablas 'estudio', 'consulta' y 'tipo_estudio' para vincular los 
--    exámenes solicitados en consulta con la descripción del catálogo.
-- 2. Filtra por la cédula del paciente a través de la consulta origen.
-- 3. Evalúa el estado del estudio mediante la condición e.estado = 'PENDIENTE' 
--    o verificando que no posea registros asociados en la tabla 'resultado'.
-- ----------------------------------------------------------------------------
SELECT 
    e.id_estudio,
    te.nombre_estudio,
    e.fecha AS fecha_solicitud,
    e.estado
FROM estudio e
JOIN consulta con ON e.id_consulta = con.id_consulta
JOIN tipo_estudio te ON e.id_tipo_estudio = te.id_tipo_estudio
WHERE con.cedula_paciente = 'V-22333444' -- Parámetro: Cédula del paciente
  AND e.estado = 'PENDIENTE';


-- ----------------------------------------------------------------------------
-- Pregunta 4: ¿Cuáles son los resultados de los exámenes de un paciente?
-- 
-- Explicación Técnica:
-- 1. Realiza un INNER JOIN en cadena desde 'resultado' pasando por 'estudio', 
--    'tipo_estudio' y 'consulta'.
-- 2. Obtiene los detalles de la muestra procesada, observaciones y la ruta 
--    al archivo multimedia del resultado digitalizado.
-- 3. Filtra la búsqueda por la cédula del paciente solicitado.
-- ----------------------------------------------------------------------------
SELECT 
    r.id_resultado,
    te.nombre_estudio,
    r.fecha AS fecha_resultado,
    r.descripcion AS observaciones_resultado,
    r.archivo_multimedia
FROM resultado r
JOIN estudio e ON r.id_estudio = e.id_estudio
JOIN tipo_estudio te ON e.id_tipo_estudio = te.id_tipo_estudio
JOIN consulta con ON e.id_consulta = con.id_consulta
WHERE con.cedula_paciente = 'V-22333444'; -- Parámetro: Cédula del paciente


-- ----------------------------------------------------------------------------
-- Pregunta 5: ¿Cuántas consultas se realizaron en un mes?
-- 
-- Explicación Técnica:
-- 1. Utiliza la función de agregación COUNT(id_consulta) para contabilizar el 
--    total de atenciones médicas registradas.
-- 2. Emplea la función TO_CHAR(fecha, 'YYYY-MM') o la extracción de rango temporal 
--    EXTRACT(MONTH FROM fecha) y EXTRACT(YEAR FROM fecha) para delimitar el período mensual.
-- ----------------------------------------------------------------------------
SELECT 
    TO_CHAR(fecha, 'YYYY-MM') AS mes_anio,
    COUNT(id_consulta) AS total_consultas
FROM consulta
WHERE TO_CHAR(fecha, 'YYYY-MM') = '2026-08' -- Parámetro: Año y Mes (YYYY-MM)
GROUP BY TO_CHAR(fecha, 'YYYY-MM');


-- ----------------------------------------------------------------------------
-- Pregunta 6: ¿Qué médicos pertenecen a una especialidad?
-- 
-- Explicación Técnica:
-- 1. Realiza una combinación de tablas (JOIN) entre 'medico', 'persona', 
--    'medico_especialidad' y 'especialidad' para asociar los datos personales 
--    del profesional con su respectiva área médica.
-- 2. Utiliza la tabla intermedia 'medico_especialidad' para resolver la 
--    relación de muchos a muchos (M:N) entre médicos y especialidades.
-- 3. Aplica la cláusula WHERE para filtrar la búsqueda por una especialidad 
--    específica mediante su clave primaria (id_especialidad).
-- ----------------------------------------------------------------------------
SELECT 
    p.cedula,
    p.nombre,
    p.apellido,
    m.carnet_medico,
    e.nombre AS especialidad
FROM medico m
JOIN persona p ON m.cedula = p.cedula
JOIN medico_especialidad me ON m.cedula = me.cedula_medico
JOIN especialidad e ON me.id_especialidad = e.id_especialidad
WHERE e.id_especialidad = 1; -- Parámetro: ID de la especialidad a consultar


-- ----------------------------------------------------------------------------
-- Pregunta 7: ¿Qué pacientes tienen citas para hoy?
-- 
-- Explicación Técnica:
-- 1. Relaciona la tabla 'cita' con 'paciente' y 'medico' (así como sus 
--    respectivas tablas de datos personales 'persona') para obtener el contexto 
--    completo del compromiso médico.
-- 2. Utiliza la función lower(c.rango_cita) para extraer el límite inferior 
--    (fecha y hora de inicio) del tipo de dato temporal TSRANGE.
-- 3. Utiliza la función DATE() combinada con CURRENT_DATE para comparar 
--    exclusivamente la fecha del día en curso con el servidor de PostgreSQL.
-- 4. Excluye las citas con estado 'CANCELADA' para mostrar únicamente la 
--    agenda operativa válida.
-- ----------------------------------------------------------------------------
SELECT 
    c.id_cita,
    c.rango_cita,
    c.consultorio,
    c.estado,
    p_pac.cedula AS cedula_paciente,
    p_pac.nombre AS nombre_paciente,
    p_pac.apellido AS apellido_paciente,
    p_med.nombre AS nombre_medico,
    p_med.apellido AS apellido_medico
FROM cita c
JOIN paciente pac ON c.cedula_paciente = pac.cedula
JOIN persona p_pac ON pac.cedula = p_pac.cedula
JOIN medico m ON c.cedula_medico = m.cedula
JOIN persona p_med ON m.cedula = p_med.cedula
WHERE DATE(lower(c.rango_cita)) = CURRENT_DATE
  AND c.estado != 'CANCELADA';


-- ----------------------------------------------------------------------------
-- Pregunta 8: ¿Cuál es el historial de consultas de un paciente?
-- 
-- Explicación Técnica:
-- 1. Unifica la tabla 'consulta' con 'paciente', 'medico' y 'persona' para 
--    desplegar la información clínica junto al médico tratante.
-- 2. Aplica un filtro directo en la cláusula WHERE por la cédula del paciente 
--    requerido.
-- 3. Ordena cronológicamente mediante ORDER BY con la cláusula DESC para que 
--    las consultas más recientes aparezcan en el primer lugar de la lista.
-- ----------------------------------------------------------------------------
SELECT 
    con.id_consulta,
    con.fecha,
    con.diagnostico,
    con.observaciones,
    con.costo,
    p_med.nombre AS nombre_medico,
    p_med.apellido AS apellido_medico
FROM consulta con
JOIN paciente pac ON con.cedula_paciente = pac.cedula
JOIN medico m ON con.cedula_medico = m.cedula
JOIN persona p_med ON m.cedula = p_med.cedula
WHERE con.cedula_paciente = 'V-22333444' -- Parámetro: Cédula del paciente
ORDER BY con.fecha DESC;


-- ----------------------------------------------------------------------------
-- Pregunta 9: ¿Qué estudios se han solicitado con mayor frecuencia?
-- 
-- Explicación Técnica:
-- 1. Realiza un INNER JOIN entre la tabla de transacciones 'estudio' (e) y la 
--    tabla catálogo 'tipo_estudio' (te) usando la clave foránea 'id_tipo_estudio'.
-- 2. Agrupa los registros mediante GROUP BY basándose en la clave primaria 
--    (te.id_tipo_estudio) y el nombre estandarizado (te.nombre_estudio), 
--    eliminando problemas de inconsistencia por texto libre.
-- 3. Emplea la función de agregación COUNT(e.id_estudio) para contabilizar el 
--    volumen total de solicitudes efectuadas por examen.
-- 4. Aplica ORDER BY DESC sobre el conteo para listar en orden decreciente 
--    los exámenes de laboratorio más demandados.
-- ----------------------------------------------------------------------------
SELECT 
    te.id_tipo_estudio,
    te.nombre_estudio,
    COUNT(e.id_estudio) AS total_solicitados
FROM estudio e
JOIN tipo_estudio te ON e.id_tipo_estudio = te.id_tipo_estudio
GROUP BY te.id_tipo_estudio, te.nombre_estudio
ORDER BY total_solicitados DESC;


-- ----------------------------------------------------------------------------
-- Pregunta 10: ¿Qué médicos tienen mayor cantidad de citas agendadas?
-- 
-- Explicación Técnica:
-- 1. Utiliza una combinación externa izquierda (LEFT JOIN) entre la tabla 
--    'medico' y 'cita'. Esto garantiza que incluso los médicos sin citas 
--    registradas aparezcan en el reporte estadístico con un conteo de cero.
-- 2. Incluye en la condición de cruce (ON) el filtro c.estado != 'CANCELADA' 
--    para asegurar que solo se contabilicen las citas activas o efectivas.
-- 3. Agrupa por el identificador del médico y contabiliza los registros válidos 
--    mediante COUNT(c.id_cita).
-- 4. Ordena el resultado de mayor a menor (DESC) según la carga global de citas.
-- ----------------------------------------------------------------------------
SELECT 
    m.cedula AS cedula_medico,
    p.nombre,
    p.apellido,
    m.carnet_medico,
    COUNT(c.id_cita) AS total_citas
FROM medico m
JOIN persona p ON m.cedula = p.cedula
LEFT JOIN cita c ON m.cedula = c.cedula_medico AND c.estado != 'CANCELADA'
GROUP BY m.cedula, p.nombre, p.apellido, m.carnet_medico
ORDER BY total_citas DESC;
